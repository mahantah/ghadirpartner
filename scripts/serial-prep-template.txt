window.prepItems=function(o){
  const shipped=shippedSet(o);
  return (o.items||[]).map((it,idx)=>{
    const p=PROD.find(p=>p.name===it.product),sr=p&&p.serial_required,serials=orderItemSerials(o,it);
    const editable=serials.filter(s=>!shipped.has(String(s).trim().toUpperCase()));
    return `<div class="card gp-prep-item"><div class="gp-prep-head"><div><b>${esc(it.product)}</b><div class="small">تعداد سفارش: ${Number(it.qty).toLocaleString('fa-IR')}</div></div>${sr?`<button type="button" class="btnlight" onclick="toggleSerialPanel(${idx})">سریال‌ها <span id="serialCount_${idx}">${serials.length}/${it.qty}</span></button>`:'<span class="small">بدون سریال اجباری</span>'}</div>${sr?`<div id="serialPanel_${idx}" class="hidden" style="margin-top:10px">
      ${partialMode(o)?'<div class="small partial-continue-note">سریال‌های ارسال‌شده قفل هستند؛ فقط باقی‌مانده را ثبت کنید.</div>':''}
      <div class="gp-prep-entry"><input id="alloc_${idx}" class="code" placeholder="اسکن سریال یا شماره کارتن از انبار"><button type="button" class="btn2" onclick="allocateWarehouseCode(${o.id},${idx})">ثبت از انبار</button></div>
      <div id="serialMessage_${idx}" class="gp-prep-message" role="status" aria-live="polite">اسکن کنید یا Enter بزنید؛ پنجره باز می‌ماند.</div><div id="serialErrors_${idx}" class="small msg" role="alert"></div>
      <details id="allocBrowser_${idx}"><summary>انتخاب از فهرست موجودی انبار</summary><div id="allocStock_${idx}" class="alloc-stock-browser"></div></details>
      <details><summary>ثبت خارج از انبار / ورود فایل</summary><div class="small" style="margin-top:7px">الگوی سریال: <span class="code">${esc(p.example||'')}</span></div><div class="toolbar" style="margin-top:7px"><input id="scan_${idx}" class="code" placeholder="سریال خارج از مخزن"><input id="imei_${idx}" class="code" inputmode="numeric" maxlength="15" placeholder="IMEI اختیاری — ۱۵ رقم"><button type="button" onclick="scanSerial(${o.id},${idx})">ثبت دستی</button></div><div class="toolbar" style="margin-top:7px"><input id="sf_${idx}" type="file" accept=".xlsx,.csv"><button type="button" class="btnlight" onclick="importOrderSerials(${o.id},${idx})">ورود Excel</button></div></details>
      <details><summary>سریال‌های ثبت‌شده <span id="serialRegisteredCount_${idx}">(${serials.length})</span></summary><div id="serialRemoveAll_${idx}">${editable.length?`<button class="danger" type="button" style="margin-top:7px" onclick="removeAllOrderSerials(${o.id},'${esc(it.product).replace(/'/g,'&#39;')}')">حذف سریال‌های ارسال‌نشده (${editable.length})</button>`:''}</div><div id="serialRegistered_${idx}" class="serial-list">${serials.length?serials.map(s=>{const iv=invBySerial(s,o.id),sent=shipped.has(String(s).trim().toUpperCase());return `<span class="serial-chip ${sent?'serial-shipped':''}"><span>${esc(s)}${iv?.imei?` <small>IMEI: ${esc(iv.imei)}</small>`:''} <small>${esc(serialSource(iv))}</small>${sent?' <b class="sent-lock">ارسال شده</b>':''}</span>${sent?'':`<button type="button" class="xbtn" onclick="removeOrderSerial(${o.id},'${esc(it.product).replace(/'/g,'&#39;')}','${esc(s)}')">×</button>`}</span>`}).join(''):'<span class="small">بدون سریال</span>'}</div></details>
    </div>`:''}</div>`;
  }).join('');
};
